Type -> strings | Numbers
Numbers -> naturalnumbers | realnumbers

// pour les trucs type 1+2+3
GeneralOperations -> number + arithmeticoperator + OperationsBis
GeneralOperationBis -> number + arithmeticoperator + OperationBis | number

// int ont un arithmetic en plus
//répétition possible?
IntOperation -> number % number

//boolean operation
//répétition possible?
BooleanOperation -> Type "and" Type | Type "or" Type |Type "==" Type
			| Type "<>" Type

//string operation
StringOperation -> strings + strings 

// get by index
GettingByIndex -> strings+lboxbracket+naturalnumber+rboxbracket
			| identifier+lboxbracket+naturalnumber+rboxbracket

// Built-in functions
NegateBoolean -> "not" + booleans
CharToInt -> "chr"+lparenthesis+naturalnumber+rparenthesis
GetLength -> "len"+lparenthesis+strings+rparenthesis | "len"+lparenthesis+identifier+rparenthesis
Floor -> "floor"+lparenthesis+realnumber+rparenthesis


// DEBUT EMBRIQUEMENT
S -> Constante + Records
	| Records
Assignement -> identifier+identifier+assignementoperator+Type+specialcharacter

//constante declaration
Constante -> "const"+Assignement

//record definition
Records -> "record" + identifier + lbracebracket + BlockRecords + rbracebrackets
BlockRecords -> identifier + type + specialcharacter + BlockRecords | epsilon (le truc pour la fin )

// Global Variables
GlobalVariables -> keywords + Assignement

// array creation
Array -> identifier+lbracebracket+rbracebracket
ArrayAssignement -> keywords + identifier + Array + assignementoperator + Array + lparenthesis+naturalnumber+rparenthesis
NEW : ArrayElement -> identifier + lboxbracket + naturalnumber + rboxbracket + assignementoperator + number + specialcharacter
NEW : ArrayFieldRecordAccess -> identifier + lboxbracket + naturalnumber + rboxbracket + identifier + assigenementOperator + number + specialcharacter
